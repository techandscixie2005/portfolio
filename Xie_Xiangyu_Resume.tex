\begin{cvsection}{Research Experience}
	\begin{cvsubsection}{Undergraduate Thesis}{Supervisor: Prof. Jun Jiang}{Fall 2025 -- Spring 2026}
		\textbf{Prediction of Room-Temperature Phosphorescence Molecular Properties Based on Equivariant Graph Neural Networks}
		\begin{itemize}
			\item \textbf{Dataset \& Targets:} Built a 3D molecular graph pipeline for about 6,000 organic RTP candidate molecules with quantum-chemistry labels for $\Delta E_{S_1-T_1}$, $SOC_{T_1 \to S_1}$, $SOC_{T_1 \to S_0}$, and $f_{\mathrm{osc},S_1}$
			\item \textbf{Model Architecture:} Designed and implemented an 8-layer EGNN\_Sparse\_Network using 21-dim atom node features, 6-dim bond edge features, 3D coordinates, and equivariant message passing
			\item \textbf{Baseline \& Training:} Compared against a character-level SMILES Transformer baseline under the same 4,800/1,050/150 train/validation/test split; applied rotation/translation augmentation only during EGNN training
			\item \textbf{Results:} Achieved test-set $R^2$ values of 0.8781, 0.6263, 0.9855, and 0.7844 for $\Delta E_{S_1-T_1}$, $SOC_{T_1 \to S_1}$, $SOC_{T_1 \to S_0}$, and $f_{\mathrm{osc},S_1}$, respectively
			\item \textbf{Conclusion:} Showed that explicit 3D geometry and equivariant inductive bias improved isolated-molecule RTP excited-state property prediction over the sequence baseline
			\item \textbf{Skills:} PyTorch, Graph Neural Networks, SE(3) Equivariance, Quantum Chemistry, Python, Linux
		\end{itemize}
	\end{cvsubsection}
